    for item in products:

        velocity = float(item["daily_sales_velocity"]) if item["daily_sales_velocity"] > 0 else 1.0
        
        days_until_empty = int(stock / velocity)
        predicted_date = today + timedelta(days=days_until_empty)
        needs_reorder = stock <= item["reorder_point"] or days_until_empty <= 7
        
        predictions.append({
            "product_id": item["id"],
            "sku": item["sku"],
            "name": item["name"],
            "current_stock": stock,
            "daily_velocity": velocity,
            "predicted_out_of_stock_date": predicted_date.strftime("%Y-%m-%d"),
            "days_remaining": days_until_empty,
            "needs_immediate_reorder": needs_reorder,
            "recommended_qty": item["recommended_reorder_qty"]
        })
        
    return {"status": "success", "data": predictions}

async function executeSimulatedApi() {
  const consoleOut = document.getElementById('console-output');
  const badge = document.getElementById('status-badge');
  const baseUrl = "http://127.0.0.1:8005"; // Updated to your active port
  
  badge.className = "text-[10px] font-mono px-1.5 py-0.5 rounded bg-amber-500/20 text-amber-400 border border-amber-500/30";
  badge.innerText = "Processing...";
  consoleOut.innerHTML = `<div class="text-indigo-400 animate-pulse">// Initiating real API request to ${baseUrl}...</div>`;

  try {
    let response;
    let logHtml = "";

    if (activeEndpoint === 'upload') {
      const sku = document.getElementById('input-sku').value || "SKU-AUDIO-9921";
      const supplier = document.getElementById('input-supplier').value || "supplier@vendor.com";
      
      const formData = new FormData();
      formData.append("sku", sku);
      formData.append("supplier_email", supplier);

      logHtml += `<div class="text-slate-400">[INFO] POST ${baseUrl}/api/upload-image</div>`;
      response = await fetch(`${baseUrl}/api/upload-image`, { method: 'POST', body: formData });
      
    } else if (activeEndpoint === 'predictions') {
      logHtml += `<div class="text-slate-400">[INFO] GET ${baseUrl}/api/inventory/predictions</div>`;
      response = await fetch(`${baseUrl}/api/inventory/predictions`);
      
    } else if (activeEndpoint === 'catalog') {
      const cat = document.getElementById('input-category')?.value;
      const url = cat ? `${baseUrl}/api/catalog?category=${encodeURIComponent(cat)}` : `${baseUrl}/api/catalog`;
      logHtml += `<div class="text-slate-400">[INFO] GET ${url}</div>`;
      response = await fetch(url);
      
    } else if (activeEndpoint === 'trigger-po') {
      const pid = document.getElementById('input-pid').value || 1;
      const qty = document.getElementById('input-qty').value || 200;
      
      logHtml += `<div class="text-slate-400">[INFO] POST ${baseUrl}/api/trigger-po</div>`;
      response = await fetch(`${baseUrl}/api/trigger-po`, {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ product_id: parseInt(pid), quantity: parseInt(qty) })
      });
    }

    const responseJson = await response.json();
    
    badge.className = "text-[10px] font-mono px-1.5 py-0.5 rounded bg-emerald-500/20 text-emerald-400 border border-emerald-500/30";
    badge.innerText = `HTTP ${response.status}`;

    logHtml += `<div class="pt-2 text-slate-200 font-semibold">Response JSON:</div>`;
    logHtml += `<pre class="text-emerald-300 font-mono text-[11px] leading-snug">${JSON.stringify(responseJson, null, 2)}</pre>`;
    
    consoleOut.innerHTML = logHtml;

  } catch (error) {
    badge.className = "text-[10px] font-mono px-1.5 py-0.5 rounded bg-red-500/20 text-red-400 border border-red-500/30";
    badge.innerText = "Error";
    consoleOut.innerHTML = `<div class="text-red-400">[ERROR] Failed to connect to ${baseUrl}. Is the FastAPI server running and allowing CORS?</div>
                            <div class="text-slate-400 text-[11px] mt-1">${error.message}</div>`;
  }
}
